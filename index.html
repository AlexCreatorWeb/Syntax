Async Loops in Python

Master concurrent execution in Python using the asyncio library. Learn how to manage tasks, gather results, and optimize I/O-bound operations.

import asyncio
import time

async def fetch_data(id: int, delay: float):
    # Simulate an I/O operation like a network request
    print(f"Task {id}: Starting download...")
    await asyncio.sleep(delay)
    print(f"Task {id}: Download complete.")
    return {"id": id, "data": f"Result of task {id}"}

async def main():
    start_time = time.time()

    # Create multiple tasks to run concurrently
    tasks = [
        fetch_data(1, 2.0),
        fetch_data(2, 1.5),
        fetch_data(3, 3.0)
    ]

    # Run tasks concurrently and wait for all to finish
    results = await asyncio.gather(*tasks)

    end_time = time.time()
    print(f"\nAll tasks completed in {end_time - start_time:.2f} seconds")
    print("Results:", results)

if __name__ == "__main__":
    asyncio.run(main())
